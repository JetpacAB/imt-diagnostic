    <script>

        const canvas = document.getElementById('imtRadar');

        const ctx = canvas.getContext('2d');

        

        const dimensions = [

            { id: 'resp', label: 'Responsibility' },

            { id: 'info', label: 'Information' },

            { id: 'inc', label: 'Incentives' },

            { id: 'adapt', label: 'Adaptation' },

            { id: 'comp', label: 'Competence' },

            { id: 'temp', label: 'Temporal' }

        ];


        function render() {

            ctx.clearRect(0, 0, 400, 400);

            const cx = 200, cy = 200, maxRadius = 140;

            

            // 1. Draw web grid lines (Levels 1 to 5)

            ctx.strokeStyle = '#e0e0e0';

            ctx.lineWidth = 1;

            for (let level = 1; level <= 5; level++) {

                ctx.beginPath();

                const r = (maxRadius / 5) * level;

                for (let i = 0; i < 6; i++) {

                    const angle = (i * Math.PI / 3) - Math.PI / 2;

                    ctx.lineTo(cx + r * Math.cos(angle), cy + r * Math.sin(angle));

                }

                ctx.closePath();

                ctx.stroke();

            }


            // 2. Draw axis lines and typography labels

            ctx.strokeStyle = '#000000';

            ctx.fillStyle = '#000000';

            ctx.font = 'bold 10px -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif';

            ctx.textAlign = 'center';

            

            dimensions.forEach((dim, i) => {

                const angle = (i * Math.PI / 3) - Math.PI / 2;

                

                // Axis line

                ctx.beginPath();

                ctx.moveTo(cx, cy);

                ctx.lineTo(cx + maxRadius * Math.cos(angle), cy + maxRadius * Math.sin(angle));

                ctx.stroke();

                

                // Text positioning offset adjustments

                const textDist = maxRadius + 20;

                let tx = cx + textDist * Math.cos(angle);

                let ty = cy + textDist * Math.sin(angle);

                if (i === 0) ty -= 5;

                if (i === 3) ty += 12;

                

                ctx.fillText(dim.label.toUpperCase(), tx, ty);

            });


            // 3. Read input scores, calculate index, identify primary bottleneck

            let totalScore = 0;

            let lowestScore = 6;

            let lowestDim = '';

            const currentPoints = [];


            dimensions.forEach((dim, i) => {

                const val = parseFloat(document.getElementById(dim.id).value);

                document.getElementById(`score-${dim.id}`).innerText = val.toFixed(1);

                

                totalScore += val;

                if (val < lowestScore) {

                    lowestScore = val;

                    lowestDim = dim.label;

                }


                const angle = (i * Math.PI / 3) - Math.PI / 2;

                const currentRadius = (maxRadius / 5) * val;

                currentPoints.push({

                    x: cx + currentRadius * Math.cos(angle),

                    y: cy + currentRadius * Math.sin(angle)

                });

            });


            // 4. Plot profile polygon

            ctx.beginPath();

            currentPoints.forEach((pt, i) => {

                if (i === 0) ctx.moveTo(pt.x, pt.y);

                else ctx.lineTo(pt.x, pt.y);

            });

            ctx.closePath();

            ctx.fillStyle = 'rgba(0, 0, 0, 0.12)';

            ctx.fill();

            ctx.lineWidth = 2;

            ctx.stroke();


            // 5. Update summary elements

            const finalIndex = (totalScore / 6).toFixed(1);

            document.getElementById('indexValue').innerText = finalIndex;

            

            document.getElementById('bottleneckOutput').innerHTML = 

                `Primary systemic bottleneck identified within <span class="bottleneck-highlight">${lowestDim}</span> (${lowestScore.toFixed(1)}). Structural interventions should prioritize this axis to unlock maturity scaling.`;

        }


        dimensions.forEach(dim => {

            document.getElementById(dim.id).addEventListener('input', render);

        });


        render();

    </script>
